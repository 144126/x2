import { describe, it, expect, vi, beforeEach } from 'vitest';
import { ChatHub } from '../hub';

class FakeReqRespPair {
	constructor(
		public request: string,
		public response: string
	) {}
}

// counts every individual storage read, so an N+1 shows up as a number that grows with N
function makeState() {
	const store = new Map<string, unknown>();
	const counts = { get: 0, list: 0 };
	return {
		counts,
		seed: (k: string, v: unknown) => store.set(k, v),
		acceptWebSocket: vi.fn(),
		getWebSockets: vi.fn(() => []),
		getTags: vi.fn(() => []),
		setWebSocketAutoResponse: vi.fn(),
		storage: {
			get: vi.fn(async (k: string) => {
				counts.get++;
				return store.get(k);
			}),
			put: vi.fn(async (k: string, v: unknown) => {
				store.set(k, v);
			}),
			delete: vi.fn(async (k: string) => {
				store.delete(k);
			}),
			list: vi.fn(async (opts?: { prefix?: string }) => {
				counts.list++;
				const m = new Map<string, unknown>();
				for (const [k, v] of store) if (!opts?.prefix || k.startsWith(opts.prefix)) m.set(k, v);
				return m;
			})
		}
	};
}

function hub_with(n: number) {
	const state = makeState();
	for (let i = 0; i < n; i++) {
		state.seed('conv:c' + i, { peer: 'p' + i, last: 1000 + i });
		state.seed('unread:c' + i, i);
	}
	const env = { CHAT_HUB: { idFromName: (x: string) => x, get: () => ({ fetch: vi.fn() }) }, SECRET: 's' };
	return { state, hub: new ChatHub(state as never, env as never) };
}

const req = (path: string) => new Request(`https://dummy${path}`);
const reads = (s: ReturnType<typeof makeState>) => s.counts.get + s.counts.list;

describe('ChatHub conversation index reads', () => {
	beforeEach(() => {
		vi.stubGlobal('WebSocketRequestResponsePair', FakeReqRespPair);
	});

	it('still returns every conversation', async () => {
		const { hub } = hub_with(5);
		const res = await hub.fetch(req('/convs'));
		expect(((await res.json()) as { convs: unknown[] }).convs).toHaveLength(5);
	});

	it('reads a constant number of times regardless of conversation count', async () => {
		const a = hub_with(2);
		await a.hub.fetch(req('/convs'));
		const b = hub_with(40);
		await b.hub.fetch(req('/convs'));
		expect(reads(b.state)).toBe(reads(a.state));
	});

	it('does not issue a per-conversation storage get', async () => {
		const { state, hub } = hub_with(40);
		await hub.fetch(req('/convs'));
		expect(state.counts.get).toBeLessThan(10);
	});

	it('reads the whole index in a handful of list calls', async () => {
		const { state, hub } = hub_with(40);
		await hub.fetch(req('/convs'));
		expect(state.counts.list).toBeLessThanOrEqual(3);
	});

	it('still reports unread counts per conversation', async () => {
		const { hub } = hub_with(3);
		const res = await hub.fetch(req('/convs'));
		const { convs } = (await res.json()) as { convs: { unread?: number }[] };
		expect(convs.some((c) => (c.unread ?? 0) > 0)).toBe(true);
	});

	it('an empty hub reads without throwing and returns nothing', async () => {
		const { hub } = hub_with(0);
		const res = await hub.fetch(req('/convs'));
		expect(((await res.json()) as { convs: unknown[] }).convs).toEqual([]);
	});
});
