import { describe, it, expect, beforeEach, vi } from 'vitest';
import { DatabaseSync } from 'node:sqlite';
import { Conv } from '../conv';

// A DurableObjectState double whose sql.exec is real SQLite, so the CREATE TABLE, the
// INSERT OR IGNORE and the ORDER BY are all genuinely exercised rather than mocked away.
function makeState() {
	const db = new DatabaseSync(':memory:');
	let alarm: number | null = null;
	return {
		alarm_at: () => alarm,
		storage: {
			sql: {
				exec: (query: string, ...bind: unknown[]) => {
					const stmt = db.prepare(query);
					if (/^\s*select/i.test(query)) {
						const rows = stmt.all(...(bind as never[]));
						return { toArray: () => rows, [Symbol.iterator]: () => rows[Symbol.iterator]() };
					}
					stmt.run(...(bind as never[]));
					return { toArray: () => [] as unknown[], [Symbol.iterator]: () => [][Symbol.iterator]() };
				}
			},
			setAlarm: vi.fn(async (t: number) => {
				alarm = t;
			}),
			getAlarm: vi.fn(async () => alarm)
		}
	};
}

function req(path: string, init?: RequestInit) {
	return new Request(`https://dummy${path}`, init);
}

const msg = (id: string, d: number, over: Record<string, unknown> = {}) => ({
	id,
	f: 'alice',
	t: 'bob',
	x: 'hello ' + id,
	d,
	...over
});

const put = (conv: Conv, m: Record<string, unknown>) =>
	conv.fetch(req('/put', { method: 'POST', body: JSON.stringify(m) }));

const page = async (conv: Conv, before?: number) => {
	const res = await conv.fetch(req('/page' + (before === undefined ? '' : `?before=${before}`)));
	return (await res.json()) as { m: { id: string; d: number; x: string; e?: number; rx?: string }[] };
};

describe('Conv', () => {
	let conv: Conv;
	beforeEach(() => {
		conv = new Conv(makeState() as never, {} as never);
	});

	it('starts empty', async () => {
		expect((await page(conv)).m).toEqual([]);
	});

	it('stores exactly one row per relayed message', async () => {
		await put(conv, msg('m1', 1000));
		const { m } = await page(conv);
		expect(m).toHaveLength(1);
		expect(m[0].id).toBe('m1');
	});

	it('a duplicate put stores no second row', async () => {
		await put(conv, msg('m1', 1000));
		await put(conv, msg('m1', 1000));
		expect((await page(conv)).m).toHaveLength(1);
	});

	it('a duplicate put does not overwrite the stored text', async () => {
		await put(conv, msg('m1', 1000));
		await put(conv, msg('m1', 1000, { x: 'clobbered' }));
		expect((await page(conv)).m[0].x).toBe('hello m1');
	});

	it('pages newest first', async () => {
		await put(conv, msg('old', 1000));
		await put(conv, msg('mid', 2000));
		await put(conv, msg('new', 3000));
		expect((await page(conv)).m.map((r) => r.id)).toEqual(['new', 'mid', 'old']);
	});

	it('before is exclusive and only returns older rows', async () => {
		await put(conv, msg('old', 1000));
		await put(conv, msg('mid', 2000));
		await put(conv, msg('new', 3000));
		expect((await page(conv, 2000)).m.map((r) => r.id)).toEqual(['old']);
	});

	it('caps a page at 50 rows', async () => {
		for (let i = 0; i < 60; i++) await put(conv, msg('m' + i, 1000 + i));
		expect((await page(conv)).m).toHaveLength(50);
	});

	it('edit replaces the text and stamps the edited time', async () => {
		await put(conv, msg('m1', 1000));
		await conv.fetch(req('/edit', { method: 'POST', body: JSON.stringify({ id: 'm1', x: 'fixed', e: 5000 }) }));
		const row = (await page(conv)).m[0];
		expect(row.x).toBe('fixed');
		expect(row.e).toBe(5000);
	});

	it('edit of an unknown id changes nothing and reports it', async () => {
		await put(conv, msg('m1', 1000));
		const res = await conv.fetch(
			req('/edit', { method: 'POST', body: JSON.stringify({ id: 'nope', x: 'fixed', e: 5000 }) })
		);
		expect(await res.json()).toEqual({ ok: 0 });
		expect((await page(conv)).m[0].x).toBe('hello m1');
	});

	it('del removes only the named row', async () => {
		await put(conv, msg('m1', 1000));
		await put(conv, msg('m2', 2000));
		await conv.fetch(req('/del', { method: 'POST', body: JSON.stringify({ id: 'm1' }) }));
		expect((await page(conv)).m.map((r) => r.id)).toEqual(['m2']);
	});

	it('react stores the reaction map as json', async () => {
		await put(conv, msg('m1', 1000));
		await conv.fetch(
			req('/react', { method: 'POST', body: JSON.stringify({ id: 'm1', rx: { '👍': ['alice'] } }) })
		);
		expect(JSON.parse((await page(conv)).m[0].rx as string)).toEqual({ '👍': ['alice'] });
	});
});
